procedure TForm2.Backup;
Var
   I: Integer;
   Searchrec: TSearchrec;
   sFrom,sTo:String;
begin
   for I := 1 to 50 do
      Begin
         sFrom:=sSave + ArrDir[i];
         sTo:=sBack + ArrDir[i];
         CopyFile(pChar(sTo),pChar(sFrom),false);
         RedOut.Lines.Add(sFrom);
         if ArrDir[i] = '' then
           Begin
              Exit;
           End;

      end;
end;

procedure TForm2.LoadDir;
Var
   myFile: Textfile;
   Text:String;
begin
   assignFile(myFile,'Bin\Dir.txt');
   Reset(myfile);
   Readln(myfile,sSave);
   Readln(myFile,sBack);
   CloseFile(myfile);
end;

procedure TForm2.Search;
Var
   srSearch:tSearchrec;
   sFirst:String;
   i:Integer;
begin
   FindFirst(sSave + '*.*',FaAnyFile,srSearch);
   I:=0;
   While sFirst <>srSearch.Name Do
      Begin
         sFirst:=srSearch.Name;
         FindNext(srSearch);
         if srSearch.Name <> '..' then
           Begin
              inc(i);
              ArrDir[i]:=srSearch.Name;
           End;
      End;
   FindClose(srSearch);
end;

procedure TForm2.Timer1Timer(Sender: TObject);
begin
   inc(iTime);
   lblTimer.Caption:='Time: ' + IntToStr(iTime);
   if iTime = 120 then
     Begin
        Search;
        Copy;
        iTime:=0;
     End;
end;
